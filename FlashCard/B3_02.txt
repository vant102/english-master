        currentIndex = Math.floor(Math.random() * data.length);
        document.getElementById('vietnameseTxt').innerText = data[currentIndex].vn;
        document.getElementById('englishTxt').innerText = data[currentIndex].en;

    function startTimer() {
        clearInterval(timer);
        timeLeft = 7;
        const bar = document.getElementById('timerBar');
        bar.style.width = '100%';
        bar.classList.remove('bg-red-500');
        bar.classList.add('bg-orange-500');

        timer = setInterval(() => {
            timeLeft--;
            bar.style.width = (timeLeft / 7 * 100) + '%';
            
            if (timeLeft <= 2) bar.classList.replace('bg-orange-500', 'bg-red-500');
            
            if (timeLeft <= 0) {
                clearInterval(timer);
                revealAnswer();

        }, 1000);

        document.getElementById('answerArea').classList.remove('hidden');

        document.getElementById('mainBtn').innerText = "CÂU TIẾP THEO";

        const msg = new SpeechSynthesisUtterance();
        msg.text = data[currentIndex].en;
        msg.lang = 'en-US';
        window.speechSynthesis.speak(msg);
    }

    // Keyboard shortcuts

            if (!document.getElementById('revealBtn').classList.contains('hidden')) {
                revealAnswer();

                nextQuestion();
            }
            e.preventDefault();
        }
    });
